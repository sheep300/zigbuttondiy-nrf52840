cmake_minimum_required(VERSION 3.20.0)
set(ZEPHYR_EXTRA_MODULES "C:/ncs/ncs-zigbee")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(zigbutton_diag_d2)
target_sources(app PRIVATE src/main.c src/d2_monitor.c)
get_property(diag_libs GLOBAL PROPERTY ZEPHYR_LIBS)
set(diag_replaced 0)
foreach(diag_lib IN LISTS diag_libs)
  get_target_property(diag_sources ${diag_lib} SOURCES)
  if(diag_sources)
    set(diag_new_sources "")
    foreach(diag_source IN LISTS diag_sources)
      if(diag_source MATCHES "(^|/)zb_nrf_platform[.]c$")
        list(APPEND diag_new_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/zb_nrf_platform_d2.c")
        math(EXPR diag_replaced "${diag_replaced} + 1")
      else()
        list(APPEND diag_new_sources "${diag_source}")
      endif()
    endforeach()
    set_property(TARGET ${diag_lib} PROPERTY SOURCES "${diag_new_sources}")
  endif()
endforeach()
if(NOT diag_replaced EQUAL 1)
  message(FATAL_ERROR "D2 must replace exactly one OSIF platform source; found ${diag_replaced}")
endif()
